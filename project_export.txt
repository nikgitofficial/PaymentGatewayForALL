"use client";

import { useState } from "react";

// "Email receipt + PDF" does three things from one click:
//   1. downloads the PAID receipt as a PDF,
//   2. copies the designed receipt email (formatted, with logo) to the clipboard,
//   3. opens a Gmail compose window with the customer's address and the subject filled in.
// In Gmail: click the message body, press Ctrl+V to paste the designed email, drag the PDF in
// (or use the paperclip), then Send. Gmail can't pre-attach files, so that step is manual.
export default function EmailReceiptButton({
  invoiceId,
  email,
  label,
  amountLabel,
  link,
  pdfLink,
  businessName,
}: {
  invoiceId: string;
  email: string;
  label: string;
  amountLabel: string;
  link: string;
  pdfLink: string;
  businessName: string;
}) {
  const [status, setStatus] = useState<"" | "rich" | "plain">("");
  const [copied, setCopied] = useState(false);

  const fileName = `${label}-receipt.pdf`;
  const downloadUrl = `/api/invoice/${invoiceId}/pdf?download=1`;
  const htmlUrl = `/api/invoice/${invoiceId}/email-html`;

  const subject = `Your receipt from ${businessName} (${label}) - PAID`;
  const plainBody =
    `Hello,\n\n` +
    `Thank you for your payment of ${amountLabel}. Your receipt for ${label}, marked as PAID, is attached as a PDF.\n\n` +
    `You can also view it online:\n${link}\n\n` +
    `PDF copy:\n${pdfLink}\n\n` +
    `Regards,\n${businessName}`;

  const gmail = (withBody: boolean) =>
    "https://mail.google.com/mail/?view=cm&fs=1" +
    `&to=${encodeURIComponent(email)}&su=${encodeURIComponent(subject)}` +
    (withBody ? `&body=${encodeURIComponent(plainBody)}` : "");

  function downloadPdf() {
    const a = document.createElement("a");
    a.href = downloadUrl;
    a.download = fileName;
    document.body.appendChild(a);
    a.click();
    a.remove();
  }

  // Designed email: copy it as rich HTML, then open an empty Gmail draft to paste it into.
  async function emailWithPdf() {
    downloadPdf();
    try {
      if (typeof ClipboardItem === "undefined" || !navigator.clipboard?.write) throw new Error("no rich clipboard");
      const html = fetch(htmlUrl, { cache: "no-store" })
        .then((r) => {
          if (!r.ok) throw new Error("fetch failed");
          return r.text();
        })
        .then((t) => new Blob([t], { type: "text/html" }));
      await navigator.clipboard.write([
        new ClipboardItem({ "text/html": html, "text/plain": new Blob([plainBody], { type: "text/plain" }) }),
      ]);
      window.open(gmail(false), "_blank", "noopener,noreferrer");
      setStatus("rich");
    } catch {
      // Clipboard blocked: fall back to a plain-text draft that already contains the message.
      window.open(gmail(true), "_blank", "noopener,noreferrer");
      setStatus("plain");
    }
  }

  function plainDraft() {
    window.open(gmail(true), "_blank", "noopener,noreferrer");
    setStatus("plain");
  }

  async function copyLink() {
    try {
      await navigator.clipboard.writeText(link);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {
      window.prompt("Copy this receipt link:", link);
    }
  }

  const btn = "rounded-lg border border-line bg-white px-3 py-1 text-xs font-semibold hover:bg-paper";

  return (
    <span className="inline-flex flex-col items-end gap-1">
      <span className="inline-flex flex-wrap items-center justify-end gap-2">
        <button type="button" onClick={emailWithPdf} className={btn}>Email receipt + PDF</button>
        <a href={downloadUrl} download={fileName} className={btn}>PDF</a>
        <button type="button" onClick={plainDraft} className={btn}>Plain draft</button>
        <button type="button" onClick={copyLink} className={btn}>{copied ? "Copied" : "Copy link"}</button>
      </span>
      {status === "rich" && (
        <span role="status" className="max-w-[18rem] text-right text-xs text-moss">
          Receipt email copied and {fileName} downloaded. In Gmail: click the message, press Ctrl+V, attach the PDF, then Send.
        </span>
      )}
      {status === "plain" && (
        <span role="status" className="max-w-[18rem] text-right text-xs text-moss">
          Plain draft opened. Attach the downloaded {fileName}, then Send.
        </span>
      )}
    </span>
  );
}

"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";

export default function IssueInvoiceForm({ defaultTotalPHP, defaultEmail = "" }: { defaultTotalPHP: number; defaultEmail?: string }) {
  const router = useRouter();
  const [email, setEmail] = useState(defaultEmail);
  const [total, setTotal] = useState(String(defaultTotalPHP));
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [done, setDone] = useState("");

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    setDone("");
    const totalPHP = Number(total.replace(/[, ₱]/g, ""));
    if (!Number.isFinite(totalPHP) || totalPHP <= 0) {
      setError("Enter a valid amount.");
      return;
    }
    if (!window.confirm(`Issue a service invoice of ₱${totalPHP.toLocaleString("en-PH", { minimumFractionDigits: 2 })} to ${email}?`)) return;
    setBusy(true);
    try {
      const res = await fetch("/api/admin/invoices", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ action: "issue", email, totalPHP }),
      });
      const d = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(d.error ?? "Failed.");
      setDone(`Invoice issued to ${email}. They'll see it after entering their email.`);
      setEmail("");
      router.refresh();
    } catch (err) {
      setError(err instanceof Error ? err.message : "Failed.");
    } finally {
      setBusy(false);
    }
  }

  const input =
    "mt-1 w-full rounded-lg border border-line bg-white px-3 py-2.5 text-sm outline-none focus:border-moss focus:ring-2 focus:ring-moss/30 disabled:opacity-60";

  return (
    <form onSubmit={submit} className="grid gap-4 sm:grid-cols-[1fr_200px_auto] sm:items-end">
      <div>
        <label htmlFor="inv-email" className="block text-sm font-semibold">Customer email</label>
        <input
          id="inv-email" type="email" required disabled={busy} value={email}
          onChange={(e) => setEmail(e.target.value)} placeholder="customer@company.com" className={input}
        />
      </div>
      <div>
        <label htmlFor="inv-total" className="block text-sm font-semibold">Total amount (PHP, VAT included)</label>
        <input
          id="inv-total" required disabled={busy} inputMode="decimal" value={total}
          onChange={(e) => setTotal(e.target.value)} className={input + " font-mono"}
        />
      </div>
      <button
        type="submit" disabled={busy}
        className="rounded-lg bg-moss px-5 py-2.5 text-sm font-semibold text-white hover:bg-moss-dark disabled:opacity-60"
      >
        {busy ? "Issuing…" : "Issue invoice"}
      </button>
      {(error || done) && (
        <p role={error ? "alert" : "status"} className={"text-sm sm:col-span-3 " + (error ? "text-red-700" : "text-moss")}>
          {error || done}
        </p>
      )}
    </form>
  );
}

"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";

export default function ReviewButtons({ claimId, expectedPHP }: { claimId: string; expectedPHP: number }) {
  const router = useRouter();
  const [busy, setBusy] = useState<"" | "confirm" | "reject">("");
  const [error, setError] = useState("");

  async function act(action: "confirm" | "reject") {
    let amountReceivedPHP: number | undefined;
    if (action === "confirm") {
      const answer = window.prompt(
        "Amount you actually received in your GCash / bank account (PHP).\nIt must be at least the invoice total.",
        String(expectedPHP)
      );
      if (answer === null) return;
      amountReceivedPHP = Number(answer.replace(/[, ₱]/g, ""));
      if (!Number.isFinite(amountReceivedPHP) || amountReceivedPHP <= 0) {
        setError("Enter a valid amount.");
        return;
      }
    }
    setBusy(action);
    setError("");
    try {
      const res = await fetch("/api/admin/review", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ claimId, action, amountReceivedPHP }),
      });
      const data = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(data.error ?? "Failed.");
      router.refresh();
    } catch (e) {
      setError(e instanceof Error ? e.message : "Failed.");
      setBusy("");
    }
  }

  return (
    <div className="flex items-center gap-2">
      <button
        onClick={() => act("confirm")} disabled={!!busy}
        className="rounded-lg bg-moss px-3 py-1.5 text-sm font-semibold text-white hover:bg-moss-dark disabled:opacity-60"
      >
        {busy === "confirm" ? "Confirming…" : "Confirm received"}
      </button>
      <button
        onClick={() => act("reject")} disabled={!!busy}
        className="rounded-lg border border-line bg-white px-3 py-1.5 text-sm font-semibold hover:bg-paper disabled:opacity-60"
      >
        {busy === "reject" ? "…" : "Reject"}
      </button>
      {error && <span role="alert" className="text-xs text-red-700">{error}</span>}
    </div>
  );
}

import { NextResponse } from "next/server";
import { z } from "zod";
import { sql } from "@/lib/db";
import { generateInvoiceItems } from "@/lib/invoice";

export const runtime = "nodejs";

// Protected by middleware.ts (HTTP Basic auth with ADMIN_PASSWORD).
// issue: creates a pending service invoice for a customer email with the amount you type (PHP, VAT included).
// void:  cancels an unpaid invoice (paid invoices can't be voided). Use it to fix a mistake, then issue a new one.
const Body = z.discriminatedUnion("action", [
  z.object({
    action: z.literal("issue"),
    email: z.string().trim().toLowerCase().email(),
    totalPHP: z.number().positive().max(100_000_000),
  }),
  z.object({ action: z.literal("void"), invoiceId: z.string().uuid() }),
]);

export async function POST(req: Request) {
  const parsed = Body.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "Enter a valid email and amount." }, { status: 400 });

  try {
    if (parsed.data.action === "void") {
      const rows = await sql`
        UPDATE invoices SET status = 'void'

    return NextResponse.json({ error: "Enter the amount you received." }, { status: 400 });

    SELECT i.total FROM payment_claims c JOIN invoices i ON i.id = c.invoice_id
    WHERE c.id = ${claimId} AND c.status = 'pending'`;
  if (!pre) return NextResponse.json({ error: "Already reviewed." }, { status: 409 });

      { status: 422 }
    );
  }

  const [claim] = await sql`
    UPDATE payment_claims SET status = 'confirmed', reviewed_at = now(), amount_received = ${received}
    WHERE id = ${claimId} AND status = 'pending'
    RETURNING id, invoice_id, method, reference`;
  if (!claim) return NextResponse.json({ error: "Already reviewed." }, { status: 409 });

  const [inv] = await sql`
    UPDATE invoices SET status = 'paid'
    WHERE id = ${claim.invoice_id} AND status <> 'paid'
    RETURNING id, lead_id, invoice_no, items, total`;

  if (inv) {
    const [lead] = await sql`SELECT email FROM leads WHERE id = ${inv.lead_id}`;

        onClick={pay}

  // While waiting, poll for the owner's confirmation.
  useEffect(() => {
    if (view !== "waiting") return;
    let stop = false;
    const tick = async () => {
      try {
        const res = await fetch(`/api/payments/status?invoiceId=${invoiceId}`, { cache: "no-store" });
        const d = await res.json();
        if (stop) return;
        if (d.invoiceStatus === "paid") router.refresh();
        else if (d.claimStatus === "rejected") setView("rejected");

    return () => { stop = true; clearInterval(t); };

    e.preventDefault();
    setBusy(true);
    setError("");

      const res = await fetch("/api/payments/claim", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ invoiceId, method: methodId, reference, senderName: sender || undefined }),
      });
      const d = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(d.error ?? "Something went wrong.");
      setView("waiting");

      setError(err instanceof Error ? err.message : "Something went wrong.");
    } finally {
      setBusy(false);
    }

  city: "", state: "", postalCode: "", country: "Philippines", countryOther: "", consent: false, website: "",
};

type Errors = Partial<Record<keyof Values, string>>;

function validate(v: Values): Errors {
  const e: Errors = {};
  if (v.businessName.trim().length < 2) e.businessName = "Enter your business name.";
  if (v.contactName.trim().length < 2) e.contactName = "Enter the contact person's name.";
  if (!/^\S+@\S+\.\S+$/.test(v.email.trim())) e.email = "Enter a valid email address.";
  if (v.phone.trim() && !/^[0-9+()\-\s.]{7,30}$/.test(v.phone.trim())) e.phone = "Enter a valid phone number.";
  if (v.tin.trim() && !/^[0-9A-Za-z][0-9A-Za-z\- ]*$/.test(v.tin.trim())) e.tin = "Use letters, numbers and dashes only.";
  if (v.addressLine.trim().length < 3) e.addressLine = "Enter your street address.";
  if (v.city.trim().length < 2) e.city = "Enter your city.";
  if (v.state.trim().length < 2) e.state = "Enter your state or province.";
  if (!/^[0-9A-Za-z][0-9A-Za-z\- ]{1,19}$/.test(v.postalCode.trim())) e.postalCode = "Enter a valid postal code.";
  if (v.country === "Other" && v.countryOther.trim().length < 2) e.countryOther = "Enter your country.";
  if (!v.consent) e.consent = "Please confirm to continue.";
  return e;
}

// Shared props for a text input bound to one field.
function inp(
  k: keyof Values,
  v: Values,
  set: <K extends keyof Values>(k: K, val: Values[K]) => void,
  errors: Errors,
  busy: boolean
) {
  return {
    id: `f-${k}`,
    name: k,
    disabled: busy,
    value: v[k] as string,
    onChange: (e: React.ChangeEvent<HTMLInputElement>) => set(k, e.target.value as Values[typeof k]),

    "aria-invalid": !!errors[k],
    "aria-describedby": errors[k] ? `e-${k}` : undefined,
  } as const;
}

// ALLOWED_EMAILS in .env: comma-separated emails and/or whole domains.
//   ALLOWED_EMAILS=jane@gmail.com,bob@company.com,@mycompany.com
// An entry starting with "@" allows everyone at that domain.
// If ALLOWED_EMAILS is empty or missing, every email is allowed.
export function isEmailAllowed(email: string) {
  const list = (process.env.ALLOWED_EMAILS ?? "")
    .split(",")
    .map((s) => s.trim().toLowerCase())
    .filter(Boolean);
  if (!list.length) return true;
  const e = email.trim().toLowerCase();
  return list.some((entry) => (entry.startsWith("@") ? e.endsWith(entry) : e === entry));
}

// Sends via Resend (https://resend.com). Needs RESEND_API_KEY and EMAIL_FROM
// (an address on a domain you verified in Resend).
export async function sendReceiptEmail(r: Receipt) {
  const key = process.env.RESEND_API_KEY;
  const from = process.env.EMAIL_FROM;
  if (!key || !from) {
    console.error("Receipt email skipped: set RESEND_API_KEY and EMAIL_FROM.");
    return;
  }
  // Add the customer's invoice history (never block the receipt if this lookup fails).
  const history = await getInvoiceHistory(r.to).catch(() => []);
  r = { ...r, history };
  const res = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" },
    body: JSON.stringify({
      from: `${brand.name} <${from}>`,
      to: [r.to],
      reply_to: brand.email || undefined,
      subject: `Your receipt from ${brand.name} (${invoiceNumber(r.invoiceNo)})`,
      html: receiptHtml(r),
    }),
  });
  if (!res.ok) console.error("Resend error:", res.status, await res.text());
}

export type HistoryRow = {
  id: string;
  invoiceNo: number;
  total: number; // centavos
  status: string;
  createdAt: Date;
};

// All service invoices billed to one email address, newest first (max 20).
// Email is matched case-insensitively through the leads table.
export async function getInvoiceHistory(email: string): Promise<HistoryRow[]> {
  const rows = await sql`
    SELECT i.id, i.invoice_no, i.total, i.status, i.created_at
    FROM invoices i
    JOIN leads l ON l.id = i.lead_id
    WHERE lower(l.email) = lower(${email})
    ORDER BY i.created_at DESC
    LIMIT 20`;
  return rows.map((r: any) => ({
    id: String(r.id),
    invoiceNo: Number(r.invoice_no),
    total: Number(r.total),
    status: String(r.status),
    createdAt: new Date(r.created_at),
  }));
}

// Splits `amount` across weights exactly (largest remainder), so parts always sum to `amount`.
function allocate(amount: number, weights: number[]) {
  const sum = weights.reduce((a, b) => a + b, 0);
  const raw = weights.map((w) => (amount * w) / sum);
  const out = raw.map(Math.floor);
  let left = amount - out.reduce((a, b) => a + b, 0);
  raw
    .map((r, i) => ({ i, frac: r - out[i] }))
    .sort((a, b) => b.frac - a.frac)
    .forEach(({ i }) => {
      if (left-- > 0) out[i]++;
    });
  return out;
}

// `totalCentavos` is the amount the owner typed when issuing an invoice (VAT included).
// Without it, the INVOICE_TOTAL_PHP env value (or the sum of the lines) is used.
function buildItems(totalCentavos?: number): InvoiceItem[] {
  const valid = INVOICE_LINES.filter((l) => Number.isFinite(l.pricePHP) && l.pricePHP > 0);
  if (!valid.length) return [];

  const withDetails = (l: (typeof valid)[number]) => (l.details?.length ? { details: l.details } : {});
  const envTotal = totalCentavos && totalCentavos > 0 ? totalCentavos : envTotalCentavos();
  let lines: InvoiceItem[];
  let tax: number;

  if (envTotal > 0) {
    // INVOICE_TOTAL_PHP is the final amount incl. tax. Remove tax, then share the rest
    // across the services in proportion to their pricePHP values.
    const subtotal = TAX_RATE > 0 ? Math.round(envTotal / (1 + TAX_RATE / 100)) : envTotal;
    const parts = allocate(subtotal, valid.map((l) => l.pricePHP));
    lines = valid
      .map((l, i) => ({ name: l.name, description: l.description, amount: parts[i], ...withDetails(l) }))
      .filter((l) => l.amount > 0);
    tax = envTotal - subtotal;
  } else {
    lines = valid.map((l) => ({
      name: l.name,
      description: l.description,
      amount: Math.round(l.pricePHP * 100),
      ...withDetails(l),
    }));
    const subtotal = lines.reduce((s, i) => s + i.amount, 0);
    tax = TAX_RATE > 0 ? Math.round((subtotal * TAX_RATE) / 100) : 0;
  }

  if (tax > 0) {
    const base = lines.reduce((s, i) => s + i.amount, 0);
    lines.push({
      name: `${TAX_LABEL} (${TAX_RATE}%)`,
      description: `${TAX_LABEL} on ${formatPHP(base)}`,
      amount: tax,
      kind: "tax",
    });
  }
  return lines;
}

// Total (incl. tax). If INVOICE_TOTAL_PHP is set in .env it wins; otherwise it is the sum of
// the lines in lib/invoice-items.ts plus tax. With no lines, falls back to a 300 PHP single item.
export function currentTotalCentavos() {
  const env = envTotalCentavos();
  if (env > 0) return env;
  const items = buildItems();
  if (items.length) return items.reduce((s, i) => s + i.amount, 0);
  return 30000;
}

export function generateInvoiceItems(totalCentavos: number): InvoiceItem[] {
  const items = buildItems(totalCentavos);
  if (items.length) return items;
  return [
    {
      name: process.env.INVOICE_ITEM_NAME || "Payment Gateway for All service",
      description: process.env.INVOICE_ITEM_DESCRIPTION || "Service payment",
      amount: totalCentavos,
    },
  ];
}

// For display: subtotal (without tax), tax amount and the tax label.
export function splitTotals(items: InvoiceItem[]) {
  const taxItem = items.find((i) => i.kind === "tax");
  const subtotal = items.filter((i) => i.kind !== "tax").reduce((s, i) => s + i.amount, 0);
  return { subtotal, tax: taxItem?.amount ?? 0, taxLabel: taxItem?.name ?? "" };
}

export const formatPHP = (centavos: number) =>
  new Intl.NumberFormat("en-PH", { style: "currency", currency: "PHP" }).format(centavos / 100);

  (process.env.INVOICE_PREFIX || "SI-") + String(no).padStart(3, "0");

// Show only the last 4 characters of sensitive numbers (account / reference numbers).
// maskTail("1234567890123") -> "*********0123"   (same length, so it still looks like a number)
export function maskTail(value: string, keep = 4): string {
  const v = value.trim();
  if (v.length <= keep) return "*".repeat(v.length);
  return "*".repeat(v.length - keep) + v.slice(-keep);
}

// Shorter form for receipts: "****0123"
export function maskShort(value: string, keep = 4): string {
  const v = value.trim();
  return "****" + v.slice(-Math.min(keep, v.length));
}

            name: i.name,

            currency: "PHP",
            quantity: 1,
          })),
          payment_method_types: ["qrph", "gcash", "paymaya", "card"],
          reference_number: opts.invoiceId,
          success_url: `${appUrl}/success`,
          cancel_url: `${appUrl}/invoice/${opts.invoiceId}`,
        },
      },
    }),
  });

  const json: any = await res.json().catch(() => null);
  if (!res.ok || !json?.data) {
    console.error("PayMongo error:", res.status, JSON.stringify(json));
    throw new Error("PayMongo request failed");
  }
  return {
    id: json.data.id as string,
    checkoutUrl: json.data.attributes.checkout_url as string,
  };
}

// Header looks like: t=1700000000,te=<testSig>,li=<liveSig>
export function verifyWebhook(rawBody: string, header: string | undefined) {
  if (!header) return false;
  const parts = Object.fromEntries(
    header.split(",").map((p) => p.split("=") as [string, string])
  );
  const expected = crypto
    .createHmac("sha256", process.env.PAYMONGO_WEBHOOK_SECRET!)
    .update(`${parts.t}.${rawBody}`)
    .digest("hex");
  const sig = process.env.PAYMONGO_SECRET_KEY!.startsWith("sk_test") ? parts.te : parts.li;
  if (!sig || sig.length !== expected.length) return false;
  return crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected));
}

export type PaidReceiptData = {

  invoiceNo: number;

  total: number; // centavos
  issuedAt: Date;
  paidAt: Date | null;
  paymentMethod?: string;
  customer: CustomerInfo;

// Loads everything a receipt (PDF or email) needs. Receipts exist only for PAID invoices.
export async function loadPaidReceipt(
  id: string
): Promise<{ kind: "notfound" } | { kind: "unpaid" } | { kind: "ok"; data: PaidReceiptData }> {
  const inv = await getInvoiceSynced(id);
  if (!inv) return { kind: "notfound" };
  if (inv.status !== "paid") return { kind: "unpaid" };

  const [claim] = await sql`
    SELECT method, reference, reviewed_at FROM payment_claims
    WHERE invoice_id = ${id} AND status = 'confirmed' ORDER BY reviewed_at DESC LIMIT 1`;
  const [pay] = claim
    ? []
    : await sql`SELECT paid_at FROM payments WHERE invoice_id = ${id} AND status = 'paid' ORDER BY paid_at DESC LIMIT 1`;
  const paidRaw = claim?.reviewed_at ?? pay?.paid_at ?? null;

  return {
    kind: "ok",
    data: {
      invoiceId: inv.id,
      invoiceNo: inv.invoice_no,
      email: inv.email,
      customer: {
        business_name: inv.business_name, contact_name: inv.contact_name, phone: inv.phone,
        address_line: inv.address_line, city: inv.city, state: inv.state,
        postal_code: inv.postal_code, country: inv.country, tin: inv.tin,
      },
      items: inv.items as InvoiceItem[],
      total: inv.total,
      issuedAt: new Date(inv.created_at),
      paidAt: paidRaw ? new Date(paidRaw) : null,
      paymentMethod: claim
        ? `${claim.method === "gcash" ? "GCash" : "Bank transfer"} · ref ${maskShort(claim.reference)}`
        : pay
          ? "PayMongo (online)"
          : undefined,
    },
  };
}

import { PDFDocument, StandardFonts, rgb, degrees, type PDFFont, type PDFPage } from "pdf-lib";
import { brand } from "./brand";
import { invoiceNumber, splitTotals, type InvoiceItem } from "./invoice";

// Builds the PAID receipt as a PDF (A4). Uses built-in PDF fonts, which have no peso sign,
// so amounts are written as "PHP 1,234.00".
type Opts = {
  invoiceNo: number;
  email: string; // bill to
  billTo?: { name?: string; contact?: string; lines: string[]; tin?: string };
  items: InvoiceItem[];
  total: number; // centavos
  issuedAt: Date;
  paidAt: Date | null;
  paymentMethod?: string;
  logoPng?: Uint8Array;
};

const money = (c: number) =>
  "PHP " + (c / 100).toLocaleString("en-PH", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
const day = (d: Date) => d.toLocaleDateString("en-PH", { dateStyle: "long" });
// Built-in fonts only cover Latin-1 + a few symbols; anything else becomes "?".
const clean = (s: string) => s.replace(/[^\x20-\x7E\u00A0-\u00FF\u2022\u2013\u2014\u2018\u2019\u201C\u201D]/g, "?");

const INK = rgb(0.08, 0.14, 0.12);
const MUTED = rgb(0.42, 0.46, 0.44);
const LINE = rgb(0.84, 0.86, 0.84);
const GREEN = rgb(0.12, 0.36, 0.29);

export async function buildReceiptPdf(o: Opts): Promise<Uint8Array> {
  const pdf = await PDFDocument.create();
  const font = await pdf.embedFont(StandardFonts.Helvetica);
  const bold = await pdf.embedFont(StandardFonts.HelveticaBold);
  const W = 595.28, H = 841.89, M = 50, R = W - M;

  let page = pdf.addPage([W, H]);
  let y = H - M;

  const text = (p: PDFPage, s: string, x: number, yy: number, size: number, f: PDFFont = font, color = INK) =>
    p.drawText(clean(s), { x, y: yy, size, font: f, color });
  const right = (s: string, yy: number, size: number, f: PDFFont = font, color = INK, edge = R) =>
    text(page, s, edge - f.widthOfTextAtSize(clean(s), size), yy, size, f, color);
  const wrap = (s: string, size: number, f: PDFFont, maxW: number) => {
    const words = clean(s).split(/\s+/);
    const lines: string[] = [];
    let cur = "";
    for (const w of words) {
      const t = cur ? cur + " " + w : w;
      if (f.widthOfTextAtSize(t, size) <= maxW) cur = t;
      else { if (cur) lines.push(cur); cur = w; }
    }
    if (cur) lines.push(cur);
    return lines;
  };
  const ensure = (need: number) => {
    if (y - need < M + 30) { page = pdf.addPage([W, H]); y = H - M; }
  };

  // ── Header
  text(page, "RECEIPT", M, y - 22, 26, bold);
  if (o.logoPng) {
    try {
      const img = await pdf.embedPng(o.logoPng);
      const h = 46, w = (img.width / img.height) * h;
      page.drawImage(img, { x: R - w, y: y - h + 4, width: w, height: h });
    } catch {}
  }
  y -= 44;

  // PAID badge
  const badge = "PAID";
  const bw = bold.widthOfTextAtSize(badge, 14) + 24;
  page.drawRectangle({ x: M, y: y - 22, width: bw, height: 24, borderColor: GREEN, borderWidth: 1.5, color: rgb(0.92, 0.97, 0.94) });
  text(page, badge, M + 12, y - 15, 14, bold, GREEN);
  y -= 46;

  // ── Details (left) and business (right)
  const rows: [string, string][] = [
    ["Invoice number", invoiceNumber(o.invoiceNo)],
    ["Date issued", day(o.issuedAt)],
    ...(o.paidAt ? ([["Date paid", day(o.paidAt)]] as [string, string][]) : []),
    ...(o.paymentMethod ? ([["Payment method", o.paymentMethod]] as [string, string][]) : []),
  ];
  const top = y;
  rows.forEach(([k, v], i) => {
    text(page, k, M, top - i * 16, 10, bold);
    text(page, v, M + 100, top - i * 16, 10);
  });
  let by = top;
  const biz = [brand.name, brand.address, brand.email, brand.tin && `TIN: ${brand.tin}`, brand.vatStatus].filter(Boolean) as string[];
  biz.forEach((l, i) => right(l, by - i * 14, i === 0 ? 11 : 10, i === 0 ? bold : font, i === 0 ? INK : MUTED));
  y = top - Math.max(rows.length * 16, biz.length * 14) - 18;

  // Compact "Bill to": name, address on one wrapped line, "Attn / TIN" on one line, email.
  const bt: { s: string; f: PDFFont; c: typeof INK }[] = [];
  if (o.billTo?.name) bt.push({ s: o.billTo.name, f: bold, c: INK });
  if (o.billTo?.lines.length) {
    wrap(o.billTo.lines.join(", "), 10, font, (R - M) * 0.7).forEach((l) => bt.push({ s: l, f: font, c: MUTED }));
  }
  const meta = [o.billTo?.contact && `Attn: ${o.billTo.contact}`, o.billTo?.tin && `TIN: ${o.billTo.tin}`].filter(Boolean);
  if (meta.length) bt.push({ s: meta.join("   |   "), f: font, c: MUTED });
  bt.push({ s: o.email, f: font, c: MUTED });
  text(page, "Bill to", M, y, 10, bold);
  bt.forEach((l, i) => text(page, l.s, M, y - 14 * (i + 1), 10, l.f, l.c));
  y -= 14 * bt.length + 28;

  // Amount paid headline
  text(page, `${money(o.total)} paid`, M, y, 18, bold);
  y -= 26;

  // ── Items
  const amtX = R;
  const descW = R - M - 110;
  page.drawLine({ start: { x: M, y: y + 4 }, end: { x: R, y: y + 4 }, thickness: 0.8, color: INK });
  text(page, "Description", M, y - 12, 9, bold);
  right("Amount", y - 12, 9, bold);
  y -= 34;

  for (const it of o.items.filter((i) => i.kind !== "tax")) {
    const desc = wrap(it.description, 9, font, descW);
    // Specs run together on wrapped lines ("a  •  b  •  c") to keep the receipt to one page.
    const det = it.details?.length ? wrap(it.details.join("  •  "), 8, font, descW) : [];
    ensure(14 + desc.length * 11 + det.length * 10 + 10);
    text(page, it.name, M, y, 10.5, bold);
    right(money(it.amount), y, 10.5, font, INK, amtX);
    y -= 13;
    desc.forEach((l) => { text(page, l, M, y, 9, font, MUTED); y -= 11; });
    det.forEach((l) => { text(page, l, M, y, 8, font, MUTED); y -= 10; });
    y -= 6;
    page.drawLine({ start: { x: M, y: y + 3 }, end: { x: R, y: y + 3 }, thickness: 0.5, color: LINE });
    y -= 8;
  }

  // ── Totals
  ensure(90);
  const { subtotal, tax, taxLabel } = splitTotals(o.items);
  if (tax > 0) {
    right("Subtotal", y, 10, font, MUTED, R - 120);
    right(money(subtotal), y, 10);
    y -= 15;
    right(taxLabel, y, 10, font, MUTED, R - 120);
    right(money(tax), y, 10);
    y -= 15;
  }
  y -= 4;
  right("Amount paid", y, 12, bold, INK, R - 120);
  right(money(o.total), y, 14, bold);
  y -= 34;

  // Stamp (light, rotated) so it reads as paid even when printed in black & white
  page.drawText("PAID", {
    x: W / 2 - 70, y: y - 40, size: 72, font: bold, color: GREEN, opacity: 0.12, rotate: degrees(12),
  });
  text(page, "Thank you for your payment.", M, y - 10, 10, font, MUTED);

  pdf.setTitle(`${invoiceNumber(o.invoiceNo)} Receipt`);
  pdf.setAuthor(brand.name);
  return pdf.save();
}

import { NextResponse, type NextRequest } from "next/server";

// Password-protects /admin and /api/admin with HTTP Basic auth.
// Set ADMIN_PASSWORD (and optionally ADMIN_USER, default "admin") in .env / Vercel.
function safeEq(a: string, b: string) {
  if (a.length !== b.length) return false;
  let r = 0;
  for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}

export function middleware(req: NextRequest) {
  const pass = process.env.ADMIN_PASSWORD;
  if (!pass) return new NextResponse("Admin is disabled: set ADMIN_PASSWORD.", { status: 503 });

  const [scheme, b64] = (req.headers.get("authorization") ?? "").split(" ");
  if (scheme === "Basic" && b64) {
    try {
      const decoded = atob(b64);
      const i = decoded.indexOf(":");
      const user = decoded.slice(0, i);
      const given = decoded.slice(i + 1);
      if (safeEq(user, process.env.ADMIN_USER || "admin") && safeEq(given, pass)) return NextResponse.next();
    } catch {}
  }
  return new NextResponse("Authentication required.", {
    status: 401,
    headers: { "WWW-Authenticate": 'Basic realm="Admin", charset="UTF-8"' },
  });
}

# Email -> (fake verify) -> PayMongo checkout (single Next.js app)

## Setup
1. Create a Neon project and copy the connection string.
2. `cp .env.example .env.local` and fill in the values.
3. `npm i && npm run db:init && npm run dev`
4. Open http://localhost:3000

## Webhook (marks payments as paid)
Expose the app (e.g. `ngrok http 3000`), set `APP_URL` to the ngrok URL, then in
PayMongo Dashboard -> Developers -> Webhooks register
`https://<your-url>/api/webhooks/paymongo` for `checkout_session.payment.paid`.
Put the webhook secret in `PAYMONGO_WEBHOOK_SECRET`.

## Notes
- Email "verification" is fake by design.
- `PRODUCT_AMOUNT` is in centavos (29900 = PHP 299.00).
- Deploys to Vercel as-is: add the same env vars and set `APP_URL` to your domain.

      items jsonb NOT NULL,
      total integer NOT NULL,
      status text NOT NULL DEFAULT 'pending',
      created_at timestamptz NOT NULL DEFAULT now()
    )`;
  await sql`
    CREATE TABLE IF NOT EXISTS payments (
      id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
      lead_id uuid NOT NULL REFERENCES leads(id),
      checkout_session_id text UNIQUE,
      amount integer NOT NULL,
      status text NOT NULL DEFAULT 'pending',
      paid_at timestamptz,
      created_at timestamptz NOT NULL DEFAULT now()
    )`;
  await sql`ALTER TABLE payments ADD COLUMN IF NOT EXISTS invoice_id uuid REFERENCES invoices(id)`;
  // Sequential invoice numbers (1, 2, 3 ...). Existing invoices are numbered automatically.
  await sql`ALTER TABLE invoices ADD COLUMN IF NOT EXISTS invoice_no integer GENERATED BY DEFAULT AS IDENTITY`;

  // Manual payments: the customer submits the GCash / bank reference, the owner confirms it.
  await sql`
    CREATE TABLE IF NOT EXISTS payment_claims (
      id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
      invoice_id uuid NOT NULL REFERENCES invoices(id),
      method text NOT NULL,
      reference text NOT NULL,
      sender_name text,
      status text NOT NULL DEFAULT 'pending',
      created_at timestamptz NOT NULL DEFAULT now(),
      reviewed_at timestamptz
    )`;
  // Amount the owner actually received (centavos), entered when confirming. For reconciliation.
  await sql`ALTER TABLE payment_claims ADD COLUMN IF NOT EXISTS amount_received integer`;
  // A reference can only be used once per method, so the same transfer can't pay two invoices.
  await sql`CREATE UNIQUE INDEX IF NOT EXISTS payment_claims_method_ref ON payment_claims (method, lower(reference))`;

  // Customer registration (/register): business details kept on the lead, used as "Bill to".
  // registered_at is set once a customer completes the form.
  await sql`ALTER TABLE leads ADD COLUMN IF NOT EXISTS business_name text`;
